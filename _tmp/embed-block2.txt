  function compute(){
    var total=0, byCat={}, byEne={A:0,M:0,B:0}, thiefCount=0, byThief={};
    var eneScore=0, eneN=0;
    var franjaEnergy={}; FRANJAS.forEach(function(f){franjaEnergy[f[0]]={sum:0,n:0};});
    CATS.forEach(function(k){byCat[k[0]]=0;});
    DIAS.forEach(function(d){
      FRANJAS.forEach(function(f){
        var c=data[d[0]][f[0]];
        var filled=c.act||c.cat||c.ene;
        if(filled) total++;
        if(c.cat) byCat[c.cat]++;
        if(c.ene){byEne[c.ene]++;var v=c.ene==="A"?3:c.ene==="M"?2:1;franjaEnergy[f[0]].sum+=v;franjaEnergy[f[0]].n++;eneScore+=v;eneN++;}
        if(c.thief){thiefCount++;var t=c.thiefType||"Sin especificar";byThief[t]=(byThief[t]||0)+1;}
      });
    });
    return {total:total,byCat:byCat,byEne:byEne,thiefCount:thiefCount,byThief:byThief,franjaEnergy:franjaEnergy,eneAvg:eneN?eneScore/eneN:0};
  }

  function bestFranja(m){
    var best=null,bestV=-1;
    FRANJAS.forEach(function(f){var fe=m.franjaEnergy[f[0]];if(fe.n){var avg=fe.sum/fe.n;if(avg>bestV){bestV=avg;best=f;}}});
    return best?{key:best[0],label:best[1],tag:best[2]}:{key:"",label:"—",tag:""};
  }
  function worstFranja(m){
    var w=null,wV=99;
    FRANJAS.forEach(function(f){var fe=m.franjaEnergy[f[0]];if(fe.n){var avg=fe.sum/fe.n;if(avg<wV){wV=avg;w=f;}}});
    return w?{label:w[1],tag:w[2]}:null;
  }
  function topKey(obj){var best=null,bv=-1;Object.keys(obj).forEach(function(k){if(obj[k]>bv){bv=obj[k];best=k;}});return best;}

  function buildInsights(m,mejor,topThief,focoPct){
    var li=[];
    if(mejor.key){
      li.push("Tu energía sube en la franja de <b>"+esc(mejor.label)+"</b> ("+esc(mejor.tag.toLowerCase())+"). Reserva ahí tu bloque de foco: es tu oro y no se gasta en correo.");
    }
    var peor=worstFranja(m);
    if(peor && peor.label!==mejor.label){
      li.push("En la franja de <b>"+esc(peor.label)+"</b> vas más floja. Es el sitio natural para gestión ligera, recados o descanso, no para lo difícil.");
    }
    if(focoPct<20){
      li.push("Solo un <b>"+focoPct+"%</b> de tu semana fue foco profundo. No es falta de tiempo: es que nadie lo estaba protegiendo. Un bloque fijo lo cambia.");
    }else if(focoPct>45){
      li.push("Dedicas mucho a foco profundo (<b>"+focoPct+"%</b>). Cuida que no se coma el descanso: los bloques también sirven para poner freno.");
    }
    if(topThief){
      var tcount=m.byThief[topThief];
      li.push("Tu mayor ladrón fue <b>"+esc(topThief)+"</b> ("+tcount+" veces). Diseña una defensa concreta: una franja sin notificaciones, una norma, un horario cerrado.");
    }
    if(m.byCat.D===0){
      li.push("No registraste ni una franja de <b>descanso</b>. Un calendario sin pausas no se sostiene; ponle bloque también a recargar.");
    }
    if(li.length<3){
      li.push("Sigue registrando unos días más: cuantas más franjas anotes, más nítido será el patrón para diseñar tu semana.");
    }
    return li.map(function(t){return "<li>"+t+"</li>";}).join("");
  }
