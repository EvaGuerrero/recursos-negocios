  var hint=SR.getElementById("hint");

  quiz.addEventListener("click",function(e){
    var opt=e.target.closest(".opt");
    if(!opt)return;
    var qi=+opt.getAttribute("data-q"), oi=+opt.getAttribute("data-o");
    picked[qi]=oi;
    answers[qi]=Q[qi].o[oi][1];
    // marcar selección visual dentro de la pregunta
    var group=opt.parentElement.querySelectorAll(".opt");
    group.forEach(function(g){g.classList.remove("sel");});
    opt.classList.add("sel");
    updateProgress();
  });

  function answeredCount(){return picked.filter(function(p){return p>=0;}).length;}
  function updateProgress(){
    var n=answeredCount();
    pfill.style.width=Math.round(n/Q.length*100)+"%";
    ptxt.textContent=n+" de "+Q.length+" respondidas";
    var done=n===Q.length;
    seeBtn.disabled=!done;
    hint.textContent=done?"¡Listo! Pulsa para ver tu cronotipo.":"Responde las "+Q.length+" preguntas para ver tu resultado.";
  }

  function compute(){
    var sc={L:0,O:0,W:0,D:0};
    answers.forEach(function(a){ if(a){ for(var k in a){ sc[k]=(sc[k]||0)+a[k]; } } });
    // ganador moderno; desempate por orden O>L>W>D (oso más común)
    var order=["O","L","W","D"];
    var best=order[0];
    order.forEach(function(k){ if(sc[k]>sc[best]) best=k; });
    // eje clásico mañana–noche (morningness)
    var me=sc.L*2 + sc.O*1 - sc.W*2;
    var trad = me>=6 ? "alondra" : (me<=-6 ? "buho" : "colibri");
    return {sc:sc,best:best,me:me,trad:trad};
  }

  function save(r){
    try{
      var full={};
      try{var raw=localStorage.getItem(STORE_KEY);if(raw)full=JSON.parse(raw)||{};}catch(e){}
      full.cronotipo={
        moderno:r.best, nombreModerno:TIPOS[r.best].name,
        tradicional:r.trad, nombreTradicional:TRAD[r.trad].name,
        scores:r.sc, morningness:r.me, fecha:new Date().toISOString()
      };
      localStorage.setItem(STORE_KEY,JSON.stringify(full));
    }catch(e){}
  }

  function diagBestFranja(){
    try{
      var raw=localStorage.getItem(STORE_KEY); if(!raw) return null;
      var o=JSON.parse(raw); var d=o&&o.diagnostico; if(!d) return null;
      var FR=[["f1","06 – 07 h"],["f2","07 – 08 h"],["f3","08 – 09 h"],["f4","09 – 10 h"],["f5","10 – 11 h"],["f6","11 – 12 h"],["f7","12 – 13 h"],["f8","13 – 14 h"],["f9","14 – 15 h"],["f10","15 – 16 h"],["f11","16 – 17 h"],["f12","17 – 18 h"],["f13","18 – 19 h"],["f14","19 – 20 h"],["f15","20 – 21 h"],["f16","21 – 22 h"]];
      var DIAS=["lun","mar","mie","jue","vie","sab","dom"];
      var best=null,bestV=-1;
      FR.forEach(function(f){var sum=0,n=0;DIAS.forEach(function(dk){var c=d[dk]&&d[dk][f[0]];if(c&&c.ene){var v=c.ene==="A"?3:c.ene==="M"?2:1;sum+=v;n++;}});if(n){var avg=sum/n;if(avg>bestV){bestV=avg;best=f[1];}}});
      return best;
    }catch(e){return null;}
  }

  function esc(s){return String(s==null?"":s).replace(/[&<>]/g,function(c){return{"&":"&amp;","<":"&lt;",">":"&gt;"}[c];});}

  function render(r){
    var tipo=r.best;
    var T=TIPOS[tipo];
    var TR=TRAD[r.trad];
    var max=Math.max.apply(null,T.curve);
    var cells=T.curve.map(function(v,i){
      var h=Math.round(v/max*100);
      return '<div class="hcell"><div class="bar"><div class="fill" style="height:'+h+'%"></div></div><div class="hr">'+HORAS[i]+'</div></div>';
    }).join("");

    var blocks=T.blocks.map(function(b){
      return '<div class="block"><div class="bt">'+esc(b[0])+'</div><div class="bd">'+b[1]+'</div></div>';
    }).join("");

    var diag=diagBestFranja();
    var cross;
    if(diag){
      cross='<span class="kick">Cruzando con tu diagnóstico</span><p>En la Guía 1, tu franja de más energía registrada fue <b>'+esc(diag)+'</b>, y tu cronotipo apunta a un pico sobre las <b>'+esc(T.peak)+'</b>. Si coinciden, ya sabes dónde va tu bloque de foco. Si no encajan del todo, suele ganar lo que registraste en tu vida real: tu cronotipo marca la tendencia, tu diagnóstico marca tu semana concreta.</p>';
    }else{
      cross='<span class="kick">Te falta una pieza</span><p>Aún no encuentro los datos de la Guía 1 en este navegador. Cuando hagas el <b>diagnóstico de una semana</b>, cruzaré tu cronotipo con tus franjas reales para afinar el bloque de foco.</p>';
    }

    SR.getElementById("result").innerHTML=
      '<div class="res-hero">'+
        '<div class="res-kick">Tu cronotipo, en las dos escalas</div>'+
        '<div class="dual">'+
          '<div class="dual-card"><span class="em">'+TR.em+'</span><div class="dl">Clásico</div><div class="dn">'+esc(TR.name)+'</div><p class="dt">'+esc(TR.tag)+'</p></div>'+
          '<div class="dual-card"><span class="em">'+T.em+'</span><div class="dl">Moderno</div><div class="dn">'+esc(T.name)+'</div><p class="dt">'+esc(T.tag)+'</p></div>'+
        '</div>'+
      '</div>'+
      '<p class="res-desc">Eres <b style="color:var(--sage-deep);font-weight:500">'+esc(TR.name)+'</b> en la escala clásica y <b style="color:var(--sage-deep);font-weight:500">'+esc(T.name)+'</b> en la moderna. '+esc(TR.desc)+'</p>'+
      '<p class="res-desc" style="margin-top:16px">'+esc(T.desc)+'</p>'+
      '<div class="curve"><h4>Tu curva de energía típica</h4><p class="sub">Así suele repartirse la energía de un '+esc(T.name)+' a lo largo del día. Cuanto más alta la barra, mejor momento. Tu pico: <b style="color:var(--sage-deep)">'+esc(T.peak)+'</b>.</p><div class="curve-scroll"><div class="curve-row">'+cells+'</div></div><p class="curve-legend">Horas de 06 a 21 h · patrón orientativo del cronotipo</p></div>'+
      '<div class="blocks"><h4>Dónde colocar tus bloques</h4><p class="sub">Tu plantilla de partida para la Guía 3, ajustada a tu reloj interno.</p><div class="block-list">'+blocks+'</div></div>'+
      '<div class="res-x">'+cross+'</div>'+
      '<div class="callout"><h2>Siguiente paso</h2><h3>Ya tienes tus horas: ahora, tu semana</h3><p>Con tu diagnóstico (Guía 1) y tu cronotipo (esta guía) guardados, en la <b>Guía 3</b> montaremos el diseño de tu semana en bloques, colocando cada cosa en su mejor hora. Guarda esta página o apunta tus cronotipos: <b>'+esc(TR.name)+'</b> (clásico) y <b>'+esc(T.name)+'</b> (moderno).</p></div>';

    SR.getElementById("result").classList.add("show");
  }

  seeBtn.addEventListener("click",function(){
    if(answeredCount()<Q.length) return;
    var r=compute();
    save(r);
    render(r);
    SR.getElementById("resultSection").scrollIntoView({behavior:"smooth",block:"start"});
  });

  updateProgress();
})();

})();
